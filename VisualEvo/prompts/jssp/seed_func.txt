def draw_v1(jobs: list, num_jobs: int, num_machines: int, out_path: str) -> None:
    import matplotlib
    matplotlib.use('Agg')
    import matplotlib.pyplot as plt
    import numpy as np

    fig_width = max(8, min(16, num_machines * 1.5))
    fig_height = max(6, min(14, num_jobs * 1.2))

    fig, ax = plt.subplots(figsize=(fig_width, fig_height), facecolor='white')
    ax.set_facecolor('white')

    # Calculate max processing time for color scaling
    max_time = max(max(op[1] for op in job) for job in jobs)

    # Color map for different machines
    colors = plt.cm.Set3(np.linspace(0, 1, num_machines))

    # Draw operation matrix
    for job_idx, job in enumerate(jobs):
        for op_idx, (machine, proc_time) in enumerate(job):
            # Position
            x = op_idx
            y = num_jobs - job_idx - 1

            # Color intensity based on processing time
            intensity = proc_time / max_time
            color = colors[machine % len(colors)]

            # Draw rectangle
            rect = plt.Rectangle((x, y), 0.9, 0.9,
                                facecolor=color,
                                edgecolor='black',
                                linewidth=1.5,
                                alpha=0.7 + 0.3 * intensity)
            ax.add_patch(rect)

            # Add text: machine and processing time
            ax.text(x + 0.45, y + 0.6, f'M{machine}',
                   ha='center', va='center',
                   fontsize=8, fontweight='bold', color='black')
            ax.text(x + 0.45, y + 0.3, f't={proc_time}',
                   ha='center', va='center',
                   fontsize=7, color='darkblue')

    # Set axis properties
    max_ops = max(len(job) for job in jobs)
    ax.set_xlim(-0.1, max_ops)
    ax.set_ylim(-0.1, num_jobs)

    # Set labels
    ax.set_xlabel('Operation Index', fontsize=10, fontweight='bold')
    ax.set_ylabel('Job Index', fontsize=10, fontweight='bold')

    # Set ticks
    ax.set_xticks(range(max_ops))
    ax.set_xticklabels([f'Op{i}' for i in range(max_ops)], fontsize=8)
    ax.set_yticks(range(num_jobs))
    ax.set_yticklabels([f'J{num_jobs - i - 1}' for i in range(num_jobs)], fontsize=8)

    ax.set_title(
        f'JSSP Instance\n'
        f'Jobs: {num_jobs}  |  Machines: {num_machines}  |  Max Time: {max_time}',
        fontsize=12, fontweight='bold', pad=10
    )

    ax.grid(True, linestyle='--', alpha=0.3)
    ax.set_aspect('equal')

    plt.tight_layout()
    plt.savefig(out_path, dpi=150, bbox_inches='tight', facecolor='white')
    plt.close(fig)
    return
